\chapter{Spells}

\section{Restoration}
\begin{Table}{l c c X}
  \TableHeader{Spell} & \TableHeader{Range} & \TableHeader{Mana Cost} & \TableHeader{Effect} \\
  Heal & Base & 0 & Target within base contact restores 1dD health. Pay an additional X mana to increase the range by X". \\
  Siphon & 6" & 0 & Make a ranged spell attack against a target within 6". Instead of dealing damage to health, reduce the target's mana by 1dD and you restore that much mana (up to your maximum mana). \\
\end{Table}

\section{Destruction}
\begin{Table}{l c c X}
  \TableHeader{Spell} & \TableHeader{Range} & \TableHeader{Mana Cost} & \TableHeader{Effect} \\
  Bolt & 12" & 1 & Make a ranged spell attack on a target within 12". Deals 1dD damage. \\
  Burst & Self & X & Deal Xd4 damage to all characters within X" radius, other than yourself. \\
\end{Table}

\section{Thaumaturgy}
\begin{Table}{l c c X}
  \TableHeader{Spell} & \TableHeader{Range} & \TableHeader{Mana Cost} & \TableHeader{Effect} \\
  Ward & Self & 0 & Gain +1 to Clash Saves until your next turn. Pay an additional X mana to target another character, and increase the range by X". \\
  Counter & Self & X & When you are targeted by a spell, you may use your reaction to pay X, where X is the amount of mana the caster spent on the spell. If you do, Clash with the caster. If you win, the spell loses all effects, otherwise the caster loses Advantage on all rolls for that spell. \\
  Focus & Self & 0 & Gain Advantage on the next Damage roll you make for a spell. If you move, this Advantage is lost. \\
\end{Table}
